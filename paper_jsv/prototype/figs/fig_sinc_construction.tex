\def\PP#1#2{({(1-#1)*(1-#2)*\gxa+#1*(1-#2)*\gxb+#1*#2*\gxc+(1-#1)*#2*\gxd+0.06*sin(180*#2)},{(1-#1)*(1-#2)*\gya+#1*(1-#2)*\gyb+#1*#2*\gyc+(1-#1)*#2*\gyd+0.07*sin(180*#1)})}

% outline of the current patch
\def\outline{\draw[structure] \PP{0}{0} \foreach \t in {0.1,0.2,...,1.0}{-- \PP{\t}{0}}
   \foreach \t in {0.1,0.2,...,1.0}{-- \PP{1}{\t}}

\def\meshlines#1{\foreach \m in {#1}{

   \draw[witness] \PP{0}{\m} \foreach \t in {0.1,0.2,...,1.0}{-- \PP{\t}{\m}};}}
