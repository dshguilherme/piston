% F18b (redesign of F18, FIG9-FIX-NOTES.md): does the deep cancellation of the exact optima survive fluid loading?
% Small multiples, one panel per cell: x = medium (in vacuo, scaled medium, air density), y = Delta L, the exact
% radiated power relative to the uniform design in the same medium (dB); higher = more power.
% figref solid: best exact-optimized design; figsurr dashed: sinc-optimized design from the uniform start;
% hollow figref diamonds: exact design re-optimized with fluid loading (30 deg, kR 3 only).
% The two lines are dodged by 0.07 of a category (exact left, sinc right) so coincident points stay visible.
% Data: data/f18b_<cell>.dat, written by tests/r0_fig_fsi_f18b.py from results/r0_revision/fsi/fsi_summary.csv
% and reopt_eval_c30_k3.csv. Drawn at 137 mm.

}

% one panel: #1 x (mm), #2 y (mm), #3 cell file stem, #4 title, #5 extra axis keys
\newcommand{\fsipanel}[5]{%
  \begin{axis}[fsiax, at={(#1 mm,#2 mm)}, #5]
    \draw (axis cs:0.5,-20) -- (axis cs:3.5,-20);
    \addplot[siline] table[x expr=\thisrow{med}+0.07, y=sinc]{data/f18b_#3.dat};
    \addplot[exline] table[x expr=\thisrow{med}-0.07, y=exact]{data/f18b_#3.dat};
  \end{axis}
  \node[paneltitle, anchor=south west] at (#1 mm,#2 mm+\ph mm+1.6mm) {#4};
}

% left gutter of one row whose axis starts at height #1 (mm): reading arrow and y label
\newcommand{\fsigutter}[1]{%
  \node[note, rotate=90, anchor=west] at (1.8mm,#1 mm) {less power};
  \node[note, rotate=90, anchor=east] at (1.8mm,#1 mm+\ph mm) {more power};
  \draw[axisline] (4.4mm,#1 mm+0.5mm) -- (4.4mm,#1 mm+\ph mm);
  \node[lab, rotate=90, anchor=north] at (6.2mm,#1 mm+0.5*\ph mm) {$\Delta L$ (dB)};
}
