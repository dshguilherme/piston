% Pilot regime figure, v2 (convention pass): (a) level above the best exact design reached by the sinc-driven
% design from the uniform start in each case (squares), with the exact optimizer's own spread (grey bar);
% higher = more power. (b) predicted against true level of the 14 sinc designs. Levels re the uniform design.

\pgfplotsset{
  axbase/.style={
    scale only axis, anchor=south west, clip=false,
    axis line style={witness}, tick style={witness},
    every tick label/.append style={note},

    major tick length=1.2mm, tick align=outside,
    axis x line*=bottom, axis y line*=left,
  },
  regax/.style={
    axbase, width=\awA mm, height=\ahA mm,

  scatax/.style={

    xticklabels={$-$80,$-$60,$-$40,$-$20,0},
    yticklabels={$-$80,$-$60,$-$40,$-$20,0},
  },
  sincfill/.style={surrcurve, only marks, mark=square*, mark size=2pt, mark options={solid, fill=figsurr}},
  sincopen/.style={surrcurve, only marks, mark=square, mark size=2pt, mark options={solid}},
  spreadbar/.style={ybar, bar width=3.2mm, fill=black!18, draw=black!45},
}

% reading arrow in the left gutter of a panel: #1 gutter x, #2 axis bottom, #3 axis height (mm)
\newcommand{\gutter}[3]{%
  \node[note, rotate=90, anchor=west] at (#1 mm+1.8mm,#2 mm) {less power};
  \node[note, rotate=90, anchor=east] at (#1 mm+1.8mm,#2 mm+#3 mm) {more power};
  \draw[axisline] (#1 mm+4.4mm,#2 mm+0.5mm) -- (#1 mm+4.4mm,#2 mm+#3 mm);
}
