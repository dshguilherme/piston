\begin{scope}[shift={(4.30,1.40)}]
  \coordinate (O) at (0,0);
  \coordinate (Ap) at (0,\Rr);
  \coordinate (PR) at ({90-\fThz}:\Rr);
  \coordinate (PL) at ({90+\fThz}:\Rr);
  % axis of symmetry (arrow up = +z) and radius to the right rim
  \draw[reference, ->, dash pattern=on 5pt off 1.5pt on 1pt off 1.5pt] (0,0) -- (0,\Rr+1.40);
  \node[lab, anchor=west] at (0.04,\Rr+1.32) {$z$};
  \draw[witness] (O) -- (PR);
  % apex-level reference line to the right gutter
  \draw[reference] (0.10,\Rr) -- ({\Ar+0.85},\Rr);
  % mesh ticks on the inside of the arc: 40 per half-arc, every 10th longer
  \foreach \k in {0,...,40} {
    \pgfmathsetmacro{\psi}{\k*\fDmer}
    \pgfmathsetmacro{\tl}{ifthenelse(mod(\k,10)==0,0.16,0.10)}
    \draw[witness, draw=black!60] ({90-\psi}:\Rr) -- ({90-\psi}:{\Rr-\tl});
    \draw[witness, draw=black!60] ({90+\psi}:\Rr) -- ({90+\psi}:{\Rr-\tl});
  }
  % the shell mid-surface
  \draw[structure] (PR) arc[start angle={90-\fThz}, end angle={90+\fThz}, radius=\Rr];
  \node[point] at (O) {};
  \node[lab, anchor=north] at (0.02,-0.08) {$O$};
  % x axis and y into the page at O
  \draw[normal] (0.07,0) -- (0.55,0);
  \node[lab, anchor=west] at (0.55,0) {$x$};
  \draw[leader] (-0.26,0) circle (0.09);
  \draw[leader] (-0.32,-0.06) -- (-0.20,0.06);
  \draw[leader] (-0.32,0.06) -- (-0.20,-0.06);
  \node[lab, anchor=south east] at (-0.20,0.10) {$y$};
  % half-angle and radius labels
  \draw[witness, ->] (90:0.80) arc[start angle=90, end angle={90-\fThz}, radius=0.80];
  \node[lab, anchor=south] at ({90-\fThz/2}:1.12) {$\theta_0$};
  \node[lab, anchor=north west] at ($(O)!0.45!(PR)+(0.04,0.0)$) {$R$};

  % upright pin supports at both rim points: triangle, apex at the rim point, hatched strip below
  \foreach \P in {PR,PL} {
    \draw[baffle] (\P) -- ($(\P)+(-0.20,-0.25)$) -- ($(\P)+(0.20,-0.25)$) -- cycle;
    \fill[hatch] ($(\P)+(-0.30,-0.25)$) rectangle ($(\P)+(0.30,-0.33)$);
    \draw[baffle] ($(\P)+(-0.30,-0.25)$) -- ($(\P)+(0.30,-0.25)$);
  }
  \draw[leader] ($(PL)+(-0.30,-0.29)$) -- (-2.00,1.76);
  \node[lab, anchor=north east, align=left] at (-1.95,1.72)
     {rim:\\ $u_x=u_y=u_z=0$,\\ rotations free};

  % height H in the right gutter
  \draw[witness] ($(PR)+(0.45,0)$) -- ($(PR)+(0.85,0)$);
  \draw[dimension] ($(PR)+(0.65,0)$) -- ($(PR)+(0.65,\Hr)$);
  \node[lab, anchor=west] at ($(PR)+(0.72,0.5*\Hr)$) {$H=\fH$ m};

  % rim radius a in the bottom gutter
  \draw[witness] ($(PR)+(0,-0.47)$) -- ($(PR|-O)+(0,-0.72)$);
  \draw[witness] (0,-0.45) -- (0,-0.72);
  \draw[dimension] (0,-0.60) -- ($(PR|-O)+(0,-0.60)$);
  \node[lab, anchor=north] at (0.5*\Ar,-0.62) {$a=R\sin\theta_0=\fA$ m};
  \node[note, anchor=north] at (0.5*\Ar,-1.12) {$R=\fR$ m, $\theta_0=\fThz^\circ$};

  % apex load along +z
  \draw[normal] (0,\Rr+0.07) -- (0,\Rr+0.80);

  % normal at 15 degrees, outer face

  \node[lab, anchor=south east, align=right] at (-0.10,\Rr+0.78)
     {$\bm n=\bm e_r$\\ (outer face)};

  % mesh note in the left interior; short leader ends on a tick of the left half-arc
  \node[note, anchor=north east, align=right] at (-0.25,2.05)
     {40 elements\\ along the\\ meridian\\ ($\fDmer^\circ$ each)};
  \draw[leader] (-1.00,2.12) -- ({90+18}:{\Rr-0.06});

  % thickness detail (not to scale), top-left; all its text directly under the circle
  \coordinate (C) at (-3.15,4.00);
  \draw[witness] (C) circle (0.80);
  \draw[reference] ($(C)+(-0.62,0)$) -- ($(C)+(0.24,0)$);
  \draw[structure] ($(C)+(-0.62,0.12)$) -- ($(C)+(0,0.12)$) -- ($(C)+(0,0.22)$) -- ($(C)+(0.16,0.22)$);
  \draw[structure] ($(C)+(-0.62,-0.12)$) -- ($(C)+(0,-0.12)$) -- ($(C)+(0,-0.22)$) -- ($(C)+(0.16,-0.22)$);
  \draw[witness] ($(C)+(0.20,0.22)$) -- ($(C)+(0.34,0.22)$);
  \draw[witness] ($(C)+(0.20,-0.22)$) -- ($(C)+(0.34,-0.22)$);
  \draw[dimension] ($(C)+(0.28,-0.22)$) -- ($(C)+(0.28,0.22)$);
  \node[lab, anchor=west] at ($(C)+(0.31,0)$) {$h_e$};
  \coordinate (DT) at ({90+24}:\Rr);
  \node[point] at (DT) {};
  \draw[leader] ($(C)+(-30:0.80)$) -- ($(DT)+(0.02,0.03)$);
  \node[note, anchor=north, align=center] at ($(C)+(0,-0.88)$)
     {detail, not to scale\\ $\fTmin\le h_e\le\fTmax$ mm,\\ mean $\le\fTmean$ mm,\\ uniform start \fTmean\ mm};

  % scale bar (4 m) and scale statement
  \begin{scope}[shift={(-4.25,-1.15)}]
    \draw[leader] (0,0) -- (\bara,0);
    \foreach \m in {0,2,4} {

  \node[note, anchor=south west] at (-4.25,-0.95) {drawn to scale};
\end{scope}
