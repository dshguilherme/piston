\def\Map#1#2{%
  \begin{axis}[mapax, at={(#2 mm,\ym mm)}, name=ax#1]
    \addplot[patch, patch type=polygon, vertex count=8, shader=faceted,
      faceted color=mapped color, witness, point meta=\thisrow{h}]
