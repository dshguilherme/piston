% F24 v2, pass rule in the level convention: error of the sinc model against its predicted level,
% with the class envelopes. x = predicted level re uniform design (negative = quieter, 0 at the right),
% y = error e = DeltaL_x - DeltaL_pred (up = louder than predicted).
% (a) design point; (b) two low-kR cells of the regime study (30 deg, kR 3; 120 deg, kR 1.5).
% Data: data/f24v2_*.dat and data/f24v2_numbers.tex (tests/r0_fig_passrule_v2.py). Drawn at 137 mm.
\documentclass[border=1mm]{standalone}
\usepackage{amsmath,amssymb,bm}
\usepackage{jsvfigs}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\input{data/f24v2_numbers.tex}

\def\ph{46}   % axis height (mm)

\pgfplotsset{
  axbase/.style={
    scale only axis, anchor=south west, clip=true,
    height=\ph mm,
    axis line style={witness}, tick style={witness},
    every tick label/.append style={note},
    label style={lab},
    major tick length=1.2mm, tick align=outside,
    axis x line*=bottom, axis y line*=left,
    xlabel={predicted level $\Delta L_{\rm pred}$ (dB)},
    ylabel style={yshift=-1mm},
  },
  sincmk/.style={surrcurve, only marks, mark=square, mark size=0.9pt, mark options={solid}},
  sincmkb/.style={surrcurve, only marks, mark=diamond, mark size=1.2pt, mark options={solid}},
  exactmk/.style={refcurve, only marks, mark=triangle, mark size=1.1pt},
  randmk/.style={refcurve, draw=black!55, only marks, mark=*, mark size=0.9pt, mark options={fill=black!55}},
  envline/.style={refcurve, draw=black},
}

% reading arrow of the y direction in the left gutter of an axis whose left edge is at x = #1 mm (bottom 10 mm)
\newcommand{\errgutter}[1]{%
  \node[note, rotate=90, anchor=west] at ({(#1-16.2)*1mm},10mm) {quieter};
  \node[note, rotate=90, anchor=east] at ({(#1-16.2)*1mm},{(10+\ph)*1mm}) {louder};
  \draw[axisline] ({(#1-13.6)*1mm},10.5mm) -- ({(#1-13.6)*1mm},{(10+\ph)*1mm});
}

\begin{tikzpicture}

\begin{axis}[axbase, at={(18mm,10mm)}, width=44mm,
    xmin=-30, xmax=3, ymin=-2.4, ymax=1.9,

    ytick={-2,-1,0,1}, yticklabels={$-$2,$-$1,0,1},
    ylabel={error $e=\Delta L_x-\Delta L_{\rm pred}$ (dB)}]
  \draw[reference] (axis cs:-30,0) -- (axis cs:3,0);
  \addplot[exactmk] table[x=dLpred, y=e]{data/f24v2_paper_exact.dat};
  \addplot[randmk] table[x=dLpred, y=e]{data/f24v2_paper_random.dat};
  \addplot[sincmk] table[x=dLpred, y=e]{data/f24v2_paper_sinc.dat};
