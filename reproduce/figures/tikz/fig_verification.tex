% Verification of the exact two-sided cap solution: (a) against boundary elements on six frozen

% Data: data/f6_bem.dat, data/f6_flat_ka{0.1,1,3,10}.dat. Drawn for the 137 mm width.
\documentclass[border=1mm]{standalone}
\usepackage{amsmath,amssymb,bm}

\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\pgfplotsset{
  vfy/.style = {
    scale only axis, axis lines=box, axis on top,
    tick align=inside, clip=false,
    every tick label/.append style={note},
    major tick length=1.5mm,
    mark size=1.7pt,
  },
  dline/.style = {normal, -, mark options={solid}, restrict x to domain=0.005:1},
  xline/.style = {reference, mark=none, restrict x to domain=-1:0.0105},
  xpt/.style = {only marks, mark=diamond*, black, mark size=2.2pt, restrict x to domain=-1:0.005},
}
% one ka series: file, marker, label y, label text, node name
\newcommand{\kaseries}[5]{%
  \addplot[xline] table[x=aR2,y=cap_over_disc] {data/f6_flat_ka#1.dat};
  \addplot[xpt] table[x=aR2,y=cap_over_disc] {data/f6_flat_ka#1.dat};
  \addplot[dline, mark=#2] table[x=aR2,y=cap_over_disc] {data/f6_flat_ka#1.dat}
     node[coordinate, pos=1] (end#5) {};
  \draw[leader] (end#5) -- (axis cs:0.1765,#3);
  \node[lab, anchor=west] at (axis cs:0.1765,#3) {#4};
}
