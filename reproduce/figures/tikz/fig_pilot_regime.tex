% Pilot regime figure: regret of the design reached from uniform per cell (a), and
% predicted against true reduction for every run (b). Data: data/f17_*.dat only. Drawn at 137 mm.
\documentclass[border=1mm]{standalone}
\usepackage{amsmath,amssymb,bm}
\usepackage{jsvfigs}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}

\pgfplotstableread{data/f17_regret.dat}\regtab
% paper-case value of the sinc run started at the exact run's end (row 7, column sincpick)
\pgfplotstablegetelem{6}{sincpick}\of\regtab
\xdef\pickval{\pgfplotsretval}

\pgfplotsset{
  axbase/.style={
    scale only axis, anchor=south west, clip=false,
    axis line style={witness}, tick style={witness},
    every tick label/.append style={note},
    label style={lab},
    major tick length=1.2mm, tick align=outside,
  },
  regax/.style={
    axbase, width=63mm, height=35mm,
    xmin=0.4, xmax=8.6, xtick=\empty, xticklabels={},
    ymin=0, ymax=65, ytick={0,20,40,60},
    axis x line*=bottom, axis y line*=left,
    ylabel={exact-power gap $s$ (dB)},
  },
  scatax/.style={
    axbase, width=50mm, height=46mm,
    xmin=-90, xmax=3, ymin=-90, ymax=3,
    xtick={-80,-60,-40,-20,0}, ytick={-80,-60,-40,-20,0},
    xticklabels={$-$80,$-$60,$-$40,$-$20,0},
    yticklabels={$-$80,$-$60,$-$40,$-$20,0},
    xlabel={predicted $\Delta L$, own objective (dB)},
    ylabel={true $\Delta L$, exact (dB)},
  },
  erpmk/.style={refcurve, draw=black!60, only marks, mark=*, mark size=1.5pt, mark options={fill=black!60}},
  erpmo/.style={refcurve, draw=black!60, only marks, mark=o, mark size=1.5pt},
  sincmk/.style={surrcurve, only marks, mark=square, mark size=1.5pt, mark options={solid}},
  exactmk/.style={refcurve, only marks, mark=triangle, mark size=2.6pt},
}

\begin{tikzpicture}

% ================================================================ panel (a)
\begin{axis}[regax, at={(10mm,14mm)}]
  % separator between the kR <= 3 and kR >= 7 cells, and the 10 dB reference
  \draw[reference] (axis cs:4.5,0) -- (axis cs:4.5,65);
  \draw[reference] (axis cs:0.4,10) -- (axis cs:7.45,10);
  \node[note, anchor=south west, align=left] at (axis cs:4.72,10.5) {10 dB: about\\the exact runs'\\own spread};
  % exact: spread of the two starts (bar) and the from-uniform value (triangle)
  \addplot[refcurve, mark=none, draw=none, error bars/.cd, y dir=both, y explicit,
      error bar style={refcurve}, error mark=none]
    table[x expr=\thisrow{idx}-0.28, y=exact,
      y error plus expr={max(\thisrow{exact},\thisrow{exactxs})-\thisrow{exact}},
      y error minus expr={\thisrow{exact}-min(\thisrow{exact},\thisrow{exactxs})}]{data/f17_regret.dat};
  \addplot[exactmk] table[x expr=\thisrow{idx}-0.28, y=exact]{data/f17_regret.dat};
  % ERP: filled when the stopping rule was met, open at the cap
  \addplot[erpmk, restrict expr to domain={\thisrow{erpcap}}{0:0}]
    table[x expr=\thisrow{idx}+0.28, y=erp]{data/f17_regret.dat};
  \addplot[erpmo, restrict expr to domain={\thisrow{erpcap}}{1:1}]
    table[x expr=\thisrow{idx}+0.28, y=erp]{data/f17_regret.dat};
  % sinc (all at the cap)
  \addplot[sincmk] table[x=idx, y=sinc]{data/f17_regret.dat};
  % paper case: sinc started at the exact run's end
  \addplot[sincmk, mark size=1.1pt] coordinates {(7,\pickval)};
  \draw[leader] (axis cs:7.12,\pickval+1.2) -- (axis cs:7.52,7.2);
  \node[note, anchor=west, align=left] at (axis cs:7.55,8.5) {from\\exact end};

\end{axis}
