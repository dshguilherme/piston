\pgfplotsset{
  designmap/.style={
    width=30mm, height=30mm, scale only axis, anchor=center,
    hide axis, axis equal image, clip=false,
    colormap={thk}{gray(0cm)=(0.93); gray(1cm)=(0.12)},
    point meta min=25, point meta max=100,
    enlargelimits=false,
  },
  patchmap/.style={patch, patch type=polygon, vertex count=8, shader=faceted,
    witness, faceted color=mapped color, point meta=\thisrow{h}},
}

% #1 data file, #2 centre x (cm), #3 centre y (cm), #4 rim radius (m), #5 optional extra
\newcommand{\DesignMap}[5]{%
  \begin{axis}[designmap, at={(#2 cm,#3 cm)}, xmin=-#4, xmax=#4, ymin=-#4, ymax=#4]
    \addplot[patchmap] table[x=x,y=y]{data/#1.dat};
    \addplot[baffle, domain=0:360, samples=121, variable=\t] ({#4*cos(t)},{#4*sin(t)});
    #5
  \end{axis}
}
