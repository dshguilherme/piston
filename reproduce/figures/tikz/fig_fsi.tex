% F18: does the deep cancellation of the exact-operator optima survive fluid loading?
% (a) power change of the best exact design and of the sinc design from uniform, three readings
% of the fluid loading; (b) exact-power gap s of the sinc design, same readings.
% Data: data/f18_fsi.dat; re-optimized points hard-coded below. Drawn at 137 mm.
\documentclass[border=1mm]{standalone}
\usepackage{amsmath,amssymb,bm}
\usepackage{jsvfigs}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}

\pgfplotstableread{data/f18_fsi.dat}\fsitab

\pgfplotsset{
  axbase/.style={
    scale only axis, anchor=south west, clip=false,
    axis line style={witness}, tick style={witness},
    every tick label/.append style={note},
    label style={lab},
    major tick length=1.2mm, tick align=outside,
    width=49mm, height=38mm,
    xmin=0.4, xmax=7.6, xtick=\empty, xticklabels={},
    axis x line*=bottom, axis y line*=left,
  },
  pwax/.style={
    axbase,
    ymin=-90, ymax=0, ytick={-80,-60,-40,-20,0},
    yticklabels={$-$80,$-$60,$-$40,$-$20,0},
    ylabel={$\Delta L$ vs uniform (dB)},
  },
  gapax/.style={
    axbase,
    ymin=0, ymax=55, ytick={0,10,20,30,40,50},
    ylabel={exact-power gap $s$ (dB)},
  },
  % marker shapes (same three for both designs); sizes differ so coincident readings stay visible
  mvacr/.style={only marks, mark=*, mark size=1.7pt, mark options={solid, fill=figref}},
  mvacs/.style={only marks, mark=*, mark size=1.7pt, mark options={solid, fill=figsurr}},
  mdev/.style={only marks, mark=o, mark size=2.2pt, mark options={solid}},
  mphy/.style={only marks, mark=x, mark size=2.3pt, mark options={solid}},
  mre/.style={refcurve, only marks, mark=diamond, mark size=2.4pt, mark options={solid}},
}

% two-line tick rows, brackets and paper-case tag under an axis whose left edge is at #1 (mm)
\def\tickrows#1{%
  \foreach \i in {0,...,6}{
    \pgfplotstablegetelem{\i}{idx}\of\fsitab \xdef\ix{\pgfplotsretval}
    \pgfplotstablegetelem{\i}{theta}\of\fsitab \xdef\th{\pgfplotsretval}
    \pgfplotstablegetelem{\i}{kR}\of\fsitab \xdef\kr{\pgfplotsretval}
    \pgfmathsetmacro{\xx}{#1+(\ix-0.4)*49/7.2}
    \node[note, anchor=north] at (\xx mm,10.4mm) {\pgfmathprintnumber[fixed,precision=0,assume math mode]{\th}$^\circ$};
    \node[note, anchor=north] at (\xx mm,7.1mm) {\pgfmathprintnumber[fixed,precision=1,zerofill,assume math mode]{\kr}};
    \draw[witness] (\xx mm,14mm) -- (\xx mm,12.8mm);
  }
  \node[note, anchor=east] at (#1 mm-1mm,9.2mm) {$\theta_0$};
  \node[note, anchor=east] at (#1 mm-1mm,5.9mm) {$kR$};
  \pgfmathsetmacro{\xA}{#1+(0.7-0.4)*49/7.2}\pgfmathsetmacro{\xB}{#1+(4.3-0.4)*49/7.2}
  \pgfmathsetmacro{\xC}{#1+(4.7-0.4)*49/7.2}\pgfmathsetmacro{\xD}{#1+(6.4-0.4)*49/7.2}
  \pgfmathsetmacro{\xAB}{(\xA+\xB)/2}\pgfmathsetmacro{\xCD}{(\xC+\xD)/2}
  \draw[witness] (\xA mm,4.4mm) -- (\xA mm,3.4mm) -- (\xB mm,3.4mm) -- (\xB mm,4.4mm);
  \node[note, anchor=north] at (\xAB mm,3.0mm) {$kR\le3$};
  \draw[witness] (\xC mm,4.4mm) -- (\xC mm,3.4mm) -- (\xD mm,3.4mm) -- (\xD mm,4.4mm);
  \node[note, anchor=north] at (\xCD mm,3.0mm) {$kR\ge7$};
  \pgfmathsetmacro{\xP}{#1+(7-0.4)*49/7.2}
  \node[note, anchor=north] at (\xP mm,3.0mm) {paper\\case};
}
